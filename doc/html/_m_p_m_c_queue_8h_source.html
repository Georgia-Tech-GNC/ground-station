69template <typename T> struct Slot {
70 ~Slot() noexcept {
71 if (turn & 1) {
72 destroy();
73 }
74 }
75
76 template <typename... Args> void construct(Args &&...args) noexcept {
77 static_assert(std::is_nothrow_constructible<T, Args &&...>::value,
78 "T must be nothrow constructible with Args&&...");
79 new (&storage) T(std::forward<Args>(args)...);
80 }
81
82 void destroy() noexcept {
83 static_assert(std::is_nothrow_destructible<T>::value,
84 "T must be nothrow destructible");
85 reinterpret_cast<T *>(&storage)->~T();
86 }
87
88 T &&move() noexcept { return reinterpret_cast<T &&>(storage); }
89
90 // Align to avoid false sharing between adjacent slots
91 alignas(hardwareInterferenceSize) std::atomic<size_t> turn = {0};
92 typename std::aligned_storage<sizeof(T), alignof(T)>::type storage;
93};

95template <typename T, typename Allocator = AlignedAllocator<Slot<T>>>
96class Queue {
97private:
98 static_assert(std::is_nothrow_copy_assignable<T>::value ||
99 std::is_nothrow_move_assignable<T>::value,
100 "T must be nothrow copy or move assignable");
101
102 static_assert(std::is_nothrow_destructible<T>::value,
103 "T must be nothrow destructible");
104
105public:
106 explicit Queue(const size_t capacity,
107 const Allocator &allocator = Allocator())
108 : capacity_(capacity), allocator_(allocator), head_(0), tail_(0) {
109 if (capacity_ < 1) {
110 throw std::invalid_argument("capacity < 1");
111 }
112 // Allocate one extra slot to prevent false sharing on the last slot
113 slots_ = allocator_.allocate(capacity_ + 1);
114 // Allocators are not required to honor alignment for over-aligned types
115 // (see http://eel.is/c++draft/allocator.requirements#10) so we verify
116 // alignment here
117 if (reinterpret_cast<size_t>(slots_) % alignof(Slot<T>) != 0) {
118 allocator_.deallocate(slots_, capacity_ + 1);
119 throw std::bad_alloc();
120 }
121 for (size_t i = 0; i < capacity_; ++i) {
122 new (&slots_[i]) Slot<T>();
123 }
124 static_assert(
125 alignof(Slot<T>) == hardwareInterferenceSize,
126 "Slot must be aligned to cache line boundary to prevent false sharing");
127 static_assert(sizeof(Slot<T>) % hardwareInterferenceSize == 0,
128 "Slot size must be a multiple of cache line size to prevent "
129 "false sharing between adjacent slots");
130 static_assert(sizeof(Queue) % hardwareInterferenceSize == 0,
131 "Queue size must be a multiple of cache line size to "
132 "prevent false sharing between adjacent queues");
133 static_assert(
134 offsetof(Queue, tail_) - offsetof(Queue, head_) ==
135 static_cast<std::ptrdiff_t>(hardwareInterferenceSize),
136 "head and tail must be a cache line apart to prevent false sharing");
137 }
138
139 ~Queue() noexcept {
140 for (size_t i = 0; i < capacity_; ++i) {
141 slots_[i].~Slot();
142 }
143 allocator_.deallocate(slots_, capacity_ + 1);
144 }
145
146 // non-copyable and non-movable
147 Queue(const Queue &) = delete;
148 Queue &operator=(const Queue &) = delete;

150 template <typename... Args> void emplace(Args &&...args) noexcept {
151 static_assert(std::is_nothrow_constructible<T, Args &&...>::value,
152 "T must be nothrow constructible with Args&&...");
153 auto const head = head_.fetch_add(1);
154 auto &slot = slots_[idx(head)];
155 while (turn(head) * 2 != slot.turn.load(std::memory_order_acquire))
156 ;
157 slot.construct(std::forward<Args>(args)...);
158 slot.turn.store(turn(head) * 2 + 1, std::memory_order_release);
159 }
160
161 template <typename... Args> bool try_emplace(Args &&...args) noexcept {
162 static_assert(std::is_nothrow_constructible<T, Args &&...>::value,
163 "T must be nothrow constructible with Args&&...");
164 auto head = head_.load(std::memory_order_acquire);
165 for (;;) {
166 auto &slot = slots_[idx(head)];
167 if (turn(head) * 2 == slot.turn.load(std::memory_order_acquire)) {
168 if (head_.compare_exchange_strong(head, head + 1)) {
169 slot.construct(std::forward<Args>(args)...);
170 slot.turn.store(turn(head) * 2 + 1, std::memory_order_release);
171 return true;
172 }
173 } else {
174 auto const prevHead = head;
175 head = head_.load(std::memory_order_acquire);
176 if (head == prevHead) {
177 return false;
178 }
179 }
180 }
181 }

210 bool pop(T &v, const std::atomic_flag& exit_flag) noexcept {
211 auto const tail = tail_.fetch_add(1);
212 auto &slot = slots_[idx(tail)];
213 bool exited = false;
214 while (turn(tail) * 2 + 1 != slot.turn.load(std::memory_order_acquire) && !(exited = exit_flag.test(std::memory_order_relaxed)))
215 ;
216 if (exited) return false;
217 v = slot.move();
218 slot.destroy();
219 slot.turn.store(turn(tail) * 2 + 2, std::memory_order_release);
220 return true;
221 }
222
223 bool try_pop(T &v) noexcept {
224 auto tail = tail_.load(std::memory_order_acquire);
225 for (;;) {
226 auto &slot = slots_[idx(tail)];
227 if (turn(tail) * 2 + 1 == slot.turn.load(std::memory_order_acquire)) {
228 if (tail_.compare_exchange_strong(tail, tail + 1)) {
229 v = slot.move();
230 slot.destroy();
231 slot.turn.store(turn(tail) * 2 + 2, std::memory_order_release);
232 return true;
233 }
234 } else {
235 auto const prevTail = tail;
236 tail = tail_.load(std::memory_order_acquire);
237 if (tail == prevTail) {
238 return false;
239 }
240 }
241 }
242 }
